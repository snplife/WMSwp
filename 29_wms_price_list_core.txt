-- WMS price list (cennik)
-- Safe/idempotent migration for existing multi-tenant schema
-- Prerequisites:
--   - public.companies
--   - public.app_users
--   - public.is_master(uuid)
--   - public.user_company_id(uuid)
--   - public.fill_company_id_from_auth_user()

create extension if not exists pgcrypto;

create table if not exists public.price_list (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  material_code text not null,
  unit text not null default 'ks',
  unit_price numeric(12,2) not null default 0,
  note text not null default '',
  created_by uuid references auth.users (id),
  updated_at timestamptz not null default now(),
  created_at timestamptz not null default now()
);

create unique index if not exists idx_price_list_company_material_unique
  on public.price_list (company_id, material_code);

create index if not exists idx_price_list_company
  on public.price_list (company_id);

create index if not exists idx_price_list_material
  on public.price_list (material_code);

create or replace function public.touch_price_list_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists trg_touch_price_list_updated_at on public.price_list;
create trigger trg_touch_price_list_updated_at
before update on public.price_list
for each row execute function public.touch_price_list_updated_at();

drop trigger if exists trg_fill_price_list_company_id on public.price_list;
create trigger trg_fill_price_list_company_id
before insert on public.price_list
for each row execute function public.fill_company_id_from_auth_user();

alter table public.price_list enable row level security;

drop policy if exists "price_list_select_auth" on public.price_list;
drop policy if exists "price_list_insert_master" on public.price_list;
drop policy if exists "price_list_update_master" on public.price_list;
drop policy if exists "price_list_delete_master" on public.price_list;
drop policy if exists "price_list_insert_user_company" on public.price_list;
drop policy if exists "price_list_update_user_company" on public.price_list;
drop policy if exists "price_list_delete_user_company" on public.price_list;

create policy "price_list_select_auth"
on public.price_list
for select
to authenticated
using (
  public.is_master(auth.uid())
  or company_id = public.user_company_id(auth.uid())
);

create policy "price_list_insert_master"
on public.price_list
for insert
to authenticated
with check (public.is_master(auth.uid()));

create policy "price_list_update_master"
on public.price_list
for update
to authenticated
using (public.is_master(auth.uid()))
with check (public.is_master(auth.uid()));

create policy "price_list_delete_master"
on public.price_list
for delete
to authenticated
using (public.is_master(auth.uid()));

create policy "price_list_insert_user_company"
on public.price_list
for insert
to authenticated
with check (company_id = public.user_company_id(auth.uid()));

create policy "price_list_update_user_company"
on public.price_list
for update
to authenticated
using (company_id = public.user_company_id(auth.uid()))
with check (company_id = public.user_company_id(auth.uid()));

create policy "price_list_delete_user_company"
on public.price_list
for delete
to authenticated
using (company_id = public.user_company_id(auth.uid()));
