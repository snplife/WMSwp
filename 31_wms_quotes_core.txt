-- WMS quotes (cenove ponuky)
-- Safe/idempotent migration for existing multi-tenant schema
-- Prerequisites:
--   - public.companies
--   - public.customers
--   - public.app_users
--   - public.is_master(uuid)
--   - public.user_company_id(uuid)
--   - public.can_manage_orders(uuid)

create extension if not exists pgcrypto;

create table if not exists public.quotes (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  customer_id uuid references public.customers (id) on delete set null,
  customer_name text not null,
  quote_number text not null,
  status text not null default 'draft' check (status in ('draft', 'sent', 'accepted', 'rejected')),
  note text not null default '',
  created_by uuid references auth.users (id),
  updated_at timestamptz not null default now(),
  created_at timestamptz not null default now()
);

create table if not exists public.quote_items (
  id uuid primary key default gen_random_uuid(),
  quote_id uuid not null references public.quotes (id) on delete cascade,
  material_code text not null,
  unit text not null default 'ks',
  quantity numeric(12,2) not null check (quantity > 0),
  unit_price numeric(12,2) not null default 0,
  purchase_price numeric(12,2) not null default 0,
  discount_percent numeric(5,2) not null default 0,
  final_unit_price numeric(12,2) not null default 0,
  line_total numeric(12,2) not null default 0,
  line_margin_total numeric(12,2) not null default 0,
  line_note text not null default '',
  created_at timestamptz not null default now()
);

create unique index if not exists idx_quotes_company_number_unique
  on public.quotes (company_id, quote_number);

create index if not exists idx_quotes_company_created_at
  on public.quotes (company_id, created_at desc);

create index if not exists idx_quote_items_quote
  on public.quote_items (quote_id);

create or replace function public.touch_quotes_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists trg_touch_quotes_updated_at on public.quotes;
create trigger trg_touch_quotes_updated_at
before update on public.quotes
for each row execute function public.touch_quotes_updated_at();

alter table public.quotes enable row level security;
alter table public.quote_items enable row level security;

drop policy if exists "quotes_select_auth" on public.quotes;
drop policy if exists "quotes_insert_auth" on public.quotes;
drop policy if exists "quotes_update_auth" on public.quotes;
drop policy if exists "quotes_delete_auth" on public.quotes;

create policy "quotes_select_auth"
on public.quotes
for select
to authenticated
using (
  public.is_master(auth.uid())
  or (
    public.can_manage_orders(auth.uid())
    and company_id = public.user_company_id(auth.uid())
  )
);

create policy "quotes_insert_auth"
on public.quotes
for insert
to authenticated
with check (
  public.is_master(auth.uid())
  or (
    public.can_manage_orders(auth.uid())
    and company_id = public.user_company_id(auth.uid())
  )
);

create policy "quotes_update_auth"
on public.quotes
for update
to authenticated
using (
  public.is_master(auth.uid())
  or (
    public.can_manage_orders(auth.uid())
    and company_id = public.user_company_id(auth.uid())
  )
)
with check (
  public.is_master(auth.uid())
  or (
    public.can_manage_orders(auth.uid())
    and company_id = public.user_company_id(auth.uid())
  )
);

create policy "quotes_delete_auth"
on public.quotes
for delete
to authenticated
using (
  public.is_master(auth.uid())
  or (
    public.can_manage_orders(auth.uid())
    and company_id = public.user_company_id(auth.uid())
  )
);

drop policy if exists "quote_items_select_auth" on public.quote_items;
drop policy if exists "quote_items_insert_auth" on public.quote_items;
drop policy if exists "quote_items_update_auth" on public.quote_items;
drop policy if exists "quote_items_delete_auth" on public.quote_items;

create policy "quote_items_select_auth"
on public.quote_items
for select
to authenticated
using (
  exists (
    select 1
    from public.quotes q
    where q.id = quote_items.quote_id
      and (
        public.is_master(auth.uid())
        or (
          public.can_manage_orders(auth.uid())
          and q.company_id = public.user_company_id(auth.uid())
        )
      )
  )
);
