-- 32_wms_customers_registry_fields.txt
-- Add structured company registry fields to public.customers.
-- Safe additive migration. Run after 27_wms_orders_core.txt.

alter table public.customers
add column if not exists ico text,
add column if not exists dic text,
add column if not exists ic_dph text;

create index if not exists idx_customers_company_ico
  on public.customers (company_id, ico);

create index if not exists idx_customers_company_dic
  on public.customers (company_id, dic);

create index if not exists idx_customers_company_ic_dph
  on public.customers (company_id, ic_dph);

update public.customers
set
  ico = coalesce(
    nullif(ico, ''),
    nullif((regexp_match(note, '(?im)(?:^|\|)\s*ICO:\s*([^|\r\n]+)'))[1], '')
  ),
  dic = coalesce(
    nullif(dic, ''),
    nullif((regexp_match(note, '(?im)(?:^|\|)\s*DIC:\s*([^|\r\n]+)'))[1], '')
  ),
  ic_dph = coalesce(
    nullif(ic_dph, ''),
    nullif((regexp_match(note, '(?im)(?:^|\|)\s*IC DPH:\s*([^|\r\n]+)'))[1], '')
  )
where
  coalesce(nullif(ico, ''), nullif(dic, ''), nullif(ic_dph, '')) is null
  and note is not null
  and note <> '';
