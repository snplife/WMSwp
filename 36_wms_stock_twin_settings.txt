-- 36_wms_stock_twin_settings.txt
-- Global digital twin layout settings shared across all companies.

create table if not exists public.stock_twin_settings (
  id integer primary key default 1 check (id = 1),
  layout jsonb not null default '{"racks":[]}'::jsonb,
  updated_at timestamptz not null default now(),
  updated_by uuid references auth.users (id)
);

insert into public.stock_twin_settings (id, layout)
values (1, '{"racks":[]}'::jsonb)
on conflict (id) do nothing;

alter table public.stock_twin_settings enable row level security;

drop policy if exists "stock_twin_settings_select_auth" on public.stock_twin_settings;
drop policy if exists "stock_twin_settings_insert_master" on public.stock_twin_settings;
drop policy if exists "stock_twin_settings_update_master" on public.stock_twin_settings;
drop policy if exists "stock_twin_settings_delete_master" on public.stock_twin_settings;

create policy "stock_twin_settings_select_auth"
on public.stock_twin_settings
for select
to authenticated
using (true);

create policy "stock_twin_settings_insert_master"
on public.stock_twin_settings
for insert
to authenticated
with check (public.is_master(auth.uid()));

create policy "stock_twin_settings_update_master"
on public.stock_twin_settings
for update
to authenticated
using (public.is_master(auth.uid()))
with check (public.is_master(auth.uid()));

create policy "stock_twin_settings_delete_master"
on public.stock_twin_settings
for delete
to authenticated
using (public.is_master(auth.uid()));
