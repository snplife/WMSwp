-- Multi-tenant WMS schema + policies (safe migration for existing DB + Android compatibility)

create extension if not exists pgcrypto;

-- 1) Core tables
create table if not exists public.companies (
  id uuid primary key default gen_random_uuid(),
  name text not null unique,
  created_at timestamptz not null default now()
);

create table if not exists public.stock (
  company_id uuid references public.companies (id) on delete cascade,
  position text not null,
  material_code text not null,
  quantity integer not null,
  constraint stock_pk primary key (company_id, position, material_code)
);

create table if not exists public.stock_history (
  event_key text primary key,
  company_id uuid references public.companies (id) on delete cascade,
  action text not null,
  position text not null,
  material_code text not null,
  note text not null default '',
  created_at_ms bigint not null
);

create table if not exists public.price_list (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  material_code text not null,
  unit text not null default 'ks',
  unit_price numeric(12,2) not null default 0,
  purchase_price numeric(12,2) not null default 0,
  note text not null default '',
  created_by uuid references auth.users (id),
  updated_at timestamptz not null default now(),
  created_at timestamptz not null default now()
);

create table if not exists public.app_users (
  user_id uuid primary key references auth.users (id) on delete cascade,
  email text not null,
  username text not null,
  role text not null check (role in ('master', 'user')),
  can_manage_orders boolean not null default false,
  company_id uuid references public.companies (id),
  created_by uuid references auth.users (id),
  db_url text,
  db_anon_key text,
  updated_at timestamptz not null default now(),
  created_at timestamptz not null default now()
);

alter table public.price_list add column if not exists purchase_price numeric(12,2) not null default 0;

alter table public.app_users add column if not exists updated_at timestamptz not null default now();

create table if not exists public.customers (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  name text not null,
  email text,
  phone text,
  address text,
  note text not null default '',
  created_by uuid references auth.users (id),
  created_at timestamptz not null default now()
);

create table if not exists public.orders (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  customer_id uuid references public.customers (id) on delete set null,
  customer_name text not null,
  order_number text not null,
  note text not null default '',
  created_by uuid references auth.users (id),
  created_at timestamptz not null default now()
);

create table if not exists public.quotes (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  customer_id uuid references public.customers (id) on delete set null,
  customer_name text not null,
  quote_number text not null,
  status text not null default 'draft' check (status in ('draft', 'sent', 'accepted', 'rejected')),
  note text not null default '',
  created_by uuid references auth.users (id),
  updated_at timestamptz not null default now(),
  created_at timestamptz not null default now()
);

create table if not exists public.quote_items (
  id uuid primary key default gen_random_uuid(),
  quote_id uuid not null references public.quotes (id) on delete cascade,
  material_code text not null,
  unit text not null default 'ks',
  quantity numeric(12,2) not null check (quantity > 0),
  unit_price numeric(12,2) not null default 0,
  purchase_price numeric(12,2) not null default 0,
  discount_percent numeric(5,2) not null default 0,
  final_unit_price numeric(12,2) not null default 0,
  line_total numeric(12,2) not null default 0,
  line_margin_total numeric(12,2) not null default 0,
  line_note text not null default '',
  created_at timestamptz not null default now()
);

create table if not exists public.order_items (
  id uuid primary key default gen_random_uuid(),
  order_id uuid not null references public.orders (id) on delete cascade,
  material_code text not null,
  position text not null,
  ordered_quantity integer not null check (ordered_quantity > 0),
  stock_quantity_snapshot integer not null default 0,
  line_note text not null default '',
  created_at timestamptz not null default now()
);

create table if not exists public.production_orders (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  production_number text not null,
  title text not null,
  status text not null default 'draft' check (status in ('draft', 'completed')),
  note text not null default '',
  created_by uuid references auth.users (id),
  completed_by uuid references auth.users (id),
  completed_at timestamptz,
  created_at timestamptz not null default now()
);

create table if not exists public.production_order_inputs (
  id uuid primary key default gen_random_uuid(),
  production_order_id uuid not null references public.production_orders (id) on delete cascade,
  material_code text not null,
  position text not null,
  required_quantity integer not null check (required_quantity > 0),
  stock_quantity_snapshot integer not null default 0,
  line_note text not null default '',
  created_at timestamptz not null default now()
);

create table if not exists public.production_order_outputs (
  id uuid primary key default gen_random_uuid(),
  production_order_id uuid not null references public.production_orders (id) on delete cascade,
  material_code text not null,
  position text not null,
  output_quantity integer not null check (output_quantity > 0),
  line_note text not null default '',
  created_at timestamptz not null default now()
);

-- 3) Normalize users
update public.app_users
set username = lower(split_part(email, '@', 1))
where (username is null or username = '')
  and email is not null;

-- 4) Default company + backfill company_id for legacy data
insert into public.companies (name)
select 'Default Company'
where not exists (select 1 from public.companies);

update public.stock
set company_id = (select id from public.companies order by created_at asc limit 1)
where company_id is null;

update public.stock_history
set company_id = (select id from public.companies order by created_at asc limit 1)
where company_id is null;

update public.app_users
set company_id = (select id from public.companies order by created_at asc limit 1)
where company_id is null
  and role = 'user';

-- Ensure stock PK includes company_id even on legacy schema
DO $$
DECLARE
  current_pk_name text;
BEGIN
  select conname into current_pk_name
  from pg_constraint
  where conrelid = 'public.stock'::regclass
    and contype = 'p';

  if current_pk_name is not null then
    execute format('alter table public.stock drop constraint %I', current_pk_name);
  end if;

  alter table public.stock
    add constraint stock_pk primary key (company_id, position, material_code);
EXCEPTION
  when duplicate_table then null;
  when duplicate_object then null;
END $$;

create or replace view public.order_list_overview
with (security_invoker = true)
as
select
  o.id,
  o.company_id,
  o.order_number,
  o.customer_name,
  o.note,
  o.created_at,
  count(oi.id)::int as item_count
from public.orders o
left join public.order_items oi on oi.order_id = o.id
group by o.id, o.company_id, o.order_number, o.customer_name, o.note, o.created_at;

create or replace function public.fill_company_id_from_auth_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if new.company_id is null then
    select au.company_id
      into new.company_id
    from public.app_users au
    where au.user_id = auth.uid()
    limit 1;
  end if;
  return new;
end;
$$;

drop trigger if exists trg_touch_app_users_updated_at on public.app_users;
create trigger trg_touch_app_users_updated_at
before update on public.app_users
for each row execute function public.touch_app_users_updated_at();

create or replace function public.is_master(uid uuid default auth.uid())
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1
    from public.app_users
    where user_id = uid
      and role = 'master'
  );
$$;

create or replace function public.user_company_id(uid uuid default auth.uid())
returns uuid
language sql
stable
security definer
set search_path = public
as $$
  select company_id
  from public.app_users
  where user_id = uid
  limit 1;
$$;

grant execute on function public.is_master(uuid) to authenticated;
grant execute on function public.user_company_id(uuid) to authenticated;

create or replace function public.can_manage_orders(uid uuid default auth.uid())
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1
    from public.app_users
    where user_id = uid
      and can_manage_orders = true
  );
$$;

grant execute on function public.can_manage_orders(uuid) to authenticated;

-- 9) Rebuild stock from stock_history events
create or replace function public.apply_stock_history_events(p_company_id uuid default null)
returns void
language plpgsql
security definer
set search_path = public
as $$
begin
  -- Rebuild target scope from history to keep stock consistent.
  if p_company_id is null then
    truncate table public.stock;

    insert into public.stock (company_id, position, material_code, quantity)
    select
      h.company_id,
      h.position,
      h.material_code,
      sum(
        case upper(coalesce(h.action, ''))
          when 'RECEIVE' then 1
          when 'ISSUE' then -1
          when 'MOVE' then 0
          when 'MOVE_ALL' then 0
          when 'ADJUST' then 0
          when 'UNDO' then 0
          else 0
        end
      )::int as quantity
    from public.stock_history h
    where h.company_id is not null
    group by h.company_id, h.position, h.material_code
    having sum(
      case upper(coalesce(h.action, ''))
        when 'RECEIVE' then 1
        when 'ISSUE' then -1
        when 'MOVE' then 0
        when 'MOVE_ALL' then 0
        when 'ADJUST' then 0
        when 'UNDO' then 0
        else 0
      end
    ) > 0;

    return;
  end if;

  delete from public.stock s
  where s.company_id = p_company_id;

  insert into public.stock (company_id, position, material_code, quantity)
  select
    h.company_id,
    h.position,
    h.material_code,
    sum(
      case upper(coalesce(h.action, ''))
        when 'RECEIVE' then 1
        when 'ISSUE' then -1
        when 'MOVE' then 0
        when 'MOVE_ALL' then 0
        when 'ADJUST' then 0
        when 'UNDO' then 0
        else 0
      end
    )::int as quantity
  from public.stock_history h
  where h.company_id = p_company_id
  group by h.company_id, h.position, h.material_code
  having sum(
    case upper(coalesce(h.action, ''))
      when 'RECEIVE' then 1
      when 'ISSUE' then -1
      when 'MOVE' then 0
      when 'MOVE_ALL' then 0
      when 'ADJUST' then 0
      when 'UNDO' then 0
      else 0
    end
  ) > 0;
end;
$$;

grant execute on function public.apply_stock_history_events(uuid) to authenticated;

-- 10) Company API keys for external stock API
create table if not exists public.company_api_keys (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  label text not null,
  key_prefix text not null,
  key_hash text not null,
  scopes text[] not null default array['stock:read']::text[],
  is_active boolean not null default true,
  expires_at timestamptz,
  last_used_at timestamptz,
  created_at timestamptz not null default now()
);

create unique index if not exists idx_company_api_keys_hash_unique
  on public.company_api_keys (key_hash);

create index if not exists idx_company_api_keys_company
  on public.company_api_keys (company_id);

create index if not exists idx_company_api_keys_company_active
  on public.company_api_keys (company_id, is_active);

alter table public.company_api_keys enable row level security;

revoke all on public.company_api_keys from anon, authenticated;
