create or replace function public.wms_rebuild_company_stock(p_company_id uuid)
returns void
language plpgsql
security definer
set search_path = public
as $$
begin
  perform public.wms_require_company_access(p_company_id);

  delete from public.stock
  where company_id = p_company_id;

  insert into public.stock (company_id, position, material_code, quantity)
  select
    h.company_id,
    h.position,
    h.material_code,
    sum(
      case h.action
        when 'RECEIVE' then 1
        when 'ISSUE' then -1
        when 'UNDO' then 0
        when 'ADJUST' then 0
        when 'MOVE_ALL' then 0
        else 0
      end
    )::int as quantity
  from public.stock_history h
  where h.company_id = p_company_id
  group by h.company_id, h.position, h.material_code
  having sum(
    case h.action
      when 'RECEIVE' then 1
      when 'ISSUE' then -1
      when 'UNDO' then 0
      when 'ADJUST' then 0
      when 'MOVE_ALL' then 0
      else 0
    end
  ) > 0;
end;
$$;

create or replace function public.wms_receive_material(
  p_company_id uuid,
  p_position text,
  p_material_code text,
  p_note text default '',
  p_created_at_ms bigint default null,
  p_event_key text default null
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_event_key text := coalesce(nullif(btrim(p_event_key), ''), gen_random_uuid()::text);
  v_created_at_ms bigint := coalesce(p_created_at_ms, floor(extract(epoch from clock_timestamp()) * 1000)::bigint);
  v_existing_position text;
begin
  perform public.wms_require_company_access(p_company_id);

  if exists (select 1 from public.stock_history where event_key = v_event_key) then
    return jsonb_build_object('ok', true, 'duplicate', true, 'event_key', v_event_key, 'action', 'RECEIVE');
  end if;

  select position
    into v_existing_position
  from public.stock
  where company_id = p_company_id
    and material_code = p_material_code
  limit 1;

  if v_existing_position is not null then
    raise exception 'Material % already exists at position %', p_material_code, v_existing_position;
  end if;

  insert into public.stock_history (event_key, company_id, action, position, material_code, note, created_at_ms)
  values (v_event_key, p_company_id, 'RECEIVE', p_position, p_material_code, coalesce(p_note, ''), v_created_at_ms);

  insert into public.stock (company_id, position, material_code, quantity)
  values (p_company_id, p_position, p_material_code, 1);

  return jsonb_build_object('ok', true, 'event_key', v_event_key, 'action', 'RECEIVE');
end;
$$;

grant execute on function public.wms_receive_material(uuid, text, text, text, bigint, text) to authenticated;

  p_created_at_ms bigint default null,
  p_event_key text default null

returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_event_key text := coalesce(nullif(btrim(p_event_key), ''), gen_random_uuid()::text);

create or replace function public.wms_move_all_position(
  p_company_id uuid,
  p_source_position text,
  p_target_position text,
  p_note text default '',
  p_created_at_ms bigint default null,
  p_event_key_base text default null
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_created_at_ms bigint := coalesce(p_created_at_ms, floor(extract(epoch from clock_timestamp()) * 1000)::bigint);
  v_event_key_base text := coalesce(nullif(btrim(p_event_key_base), ''), gen_random_uuid()::text);
  v_row record;
  v_event_key text;
  v_materials integer := 0;
begin
  perform public.wms_require_company_access(p_company_id);

  if p_source_position = p_target_position then
    raise exception 'Source and target position must be different';
  end if;

  for v_row in
    select material_code, quantity
    from public.stock
    where company_id = p_company_id
      and position = p_source_position
    for update
  loop
    if coalesce(v_row.quantity, 0) <= 0 then
      continue;
    end if;

    if exists (
      select 1
      from public.stock
      where company_id = p_company_id
        and position = p_target_position
        and material_code = v_row.material_code
    ) then
      raise exception 'Material % already exists at target position %', v_row.material_code, p_target_position;
    end if;

    v_event_key := v_event_key_base || ':' || v_row.material_code;

    if not exists (select 1 from public.stock_history where event_key = v_event_key) then
      insert into public.stock_history (event_key, company_id, action, position, material_code, note, created_at_ms)
      values (
        v_event_key,
        p_company_id,
        'MOVE_ALL',
        p_source_position,
        v_row.material_code,
        coalesce(p_note, '') || ' | TARGET:' || p_target_position,
        v_created_at_ms
      );
    end if;

    insert into public.stock (company_id, position, material_code, quantity)
    values (p_company_id, p_target_position, v_row.material_code, v_row.quantity)
    on conflict (company_id, position, material_code)
    do nothing;

    v_materials := v_materials + 1;
  end loop;

  delete from public.stock
  where company_id = p_company_id
    and position = p_source_position;

  return jsonb_build_object(
    'ok', true,
    'action', 'MOVE_ALL',
    'event_key_base', v_event_key_base,
    'materials', v_materials
  );
end;
$$;

grant execute on function public.wms_move_all_position(uuid, text, text, text, bigint, text) to authenticated;

-- Adds app_role + company_id claims to JWT from public.app_users.
-- After running, enable this function in Supabase Dashboard:
-- Authentication -> Hooks -> Custom access token hook -> public.custom_access_token_hook

create or replace function public.custom_access_token_hook(event jsonb)
returns jsonb
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  uid uuid;
  claims jsonb;
  v_role text;
  v_company_id uuid;
begin
  uid := nullif(event->>'user_id', '')::uuid;
  claims := coalesce(event->'claims', '{}'::jsonb);

  if uid is null then
    return event;
  end if;

  select role, company_id
    into v_role, v_company_id
  from public.app_users
  where user_id = uid;

  claims := jsonb_set(claims, '{app_role}', to_jsonb(coalesce(v_role, 'user')), true);

  if v_company_id is not null then
    claims := jsonb_set(claims, '{company_id}', to_jsonb(v_company_id::text), true);
  else
    claims := claims - 'company_id';
  end if;

  event := jsonb_set(event, '{claims}', claims, true);
  return event;
end;
$$;

create or replace function public.wms_upsert_user_profile(
  p_user_id uuid,
  p_email text,
  p_username text,
  p_role text default 'user',
  p_can_manage_orders boolean default false,
  p_company_id uuid default null,
  p_db_url text default null,
  p_db_anon_key text default null,
  p_created_by uuid default auth.uid()
)
returns public.app_users
language plpgsql
security definer
set search_path = public
as $$
declare
  v_role text := lower(coalesce(p_role, 'user'));
  v_row public.app_users;
begin
  if not public.is_master(auth.uid()) and not public.can_bootstrap_master(p_user_id, v_role) then
    raise exception 'Only master can upsert user profiles';
  end if;

  if v_role not in ('master', 'user') then
    raise exception 'Invalid role: %', v_role;
  end if;

  if v_role = 'user' and p_company_id is null then
    raise exception 'company_id is required for user role';
  end if;

  insert into public.app_users (
    user_id, email, username, role, can_manage_orders, company_id, db_url, db_anon_key, created_by
  )
  values (
    p_user_id,
    lower(coalesce(p_email, '')),
    lower(coalesce(p_username, split_part(coalesce(p_email, ''), '@', 1))),
    v_role,
    case when v_role = 'master' then true else coalesce(p_can_manage_orders, false) end,
    case when v_role = 'master' then null else p_company_id end,
    p_db_url,
    p_db_anon_key,
    p_created_by
  )
  on conflict (user_id)
  do update set
    email = excluded.email,
    username = excluded.username,
    role = excluded.role,
    can_manage_orders = excluded.can_manage_orders,
    company_id = excluded.company_id,
    db_url = coalesce(excluded.db_url, public.app_users.db_url),
    db_anon_key = coalesce(excluded.db_anon_key, public.app_users.db_anon_key),
    updated_at = now()
  returning * into v_row;

  return v_row;
end;

create or replace function public.claim_email_alert_batch(
  p_limit integer default 20,
  p_max_attempts integer default 5,
  p_lock_minutes integer default 10
)
returns table (
  id uuid,
  subscription_id uuid,
  company_id uuid,
  event_key text,
  action text,
  material_code text,
  item_position text,
  note text,
  created_at_ms bigint,

  attempts integer,
  payload jsonb,
  queued_at timestamptz

language plpgsql
security definer
set search_path = public
as $$
begin

  with candidates as (
    select q.id
    from public.email_alert_queue q

      and (
        q.locked_at is null
        or q.locked_at < now() - make_interval(mins => greatest(coalesce(p_lock_minutes, 10), 1))
      )

    limit greatest(coalesce(p_limit, 20), 1)
  ),
  claimed as (
    update public.email_alert_queue q
    set status = 'processing',
        locked_at = now(),
        attempts = q.attempts + 1
    from candidates c
    where q.id = c.id
    returning
      q.id,
      q.subscription_id,
      q.company_id,
      q.event_key,
      q.action,
      q.material_code,
      q.position as item_position,
      q.note,
      q.created_at_ms,
      q.recipient_email,
      q.attempts,
      q.payload,
      q.queued_at
  )

  from claimed;

$$;

create or replace function public.wms_rebuild_company_stock(p_company_id uuid)
returns void
language plpgsql
security definer
set search_path = public
as $$
begin
  perform public.wms_require_company_access(p_company_id);

  delete from public.stock
  where company_id = p_company_id;

  insert into public.stock (company_id, position, material_code, quantity, expiry_date)
  select
    h.company_id,
    h.position,
    h.material_code,
    sum(
      case h.action
        when 'RECEIVE' then 1
        when 'ISSUE' then -1
        when 'UNDO' then 0
        when 'ADJUST' then 0
        when 'MOVE_ALL' then 0
        else 0
      end
    )::int as quantity,
    coalesce(
      max(nullif(h.expiry_date, '')) filter (where h.action in ('RECEIVE', 'UNDO')),
      ''
    ) as expiry_date
  from public.stock_history h
  where h.company_id = p_company_id
  group by h.company_id, h.position, h.material_code
  having sum(
    case h.action
      when 'RECEIVE' then 1
      when 'ISSUE' then -1
      when 'UNDO' then 0
      when 'ADJUST' then 0
      when 'MOVE_ALL' then 0
      else 0
    end
  ) > 0;
end;
$$;

  v_row record;
  v_event_key text;
  v_materials integer := 0;
begin
  perform public.wms_require_company_access(p_company_id);

  if p_source_position = p_target_position then
    raise exception 'Source and target position must be different';
  end if;

  for v_row in
    select material_code, quantity, expiry_date
    from public.stock
    where company_id = p_company_id
      and position = p_source_position
    for update
  loop
    if coalesce(v_row.quantity, 0) <= 0 then
      continue;
    end if;

    if exists (
      select 1
      from public.stock
      where company_id = p_company_id
        and position = p_target_position
        and material_code = v_row.material_code
    ) then
      raise exception 'Material % already exists at target position %', v_row.material_code, p_target_position;
    end if;

    v_event_key := v_event_key_base || ':' || v_row.material_code;

    if not exists (select 1 from public.stock_history where event_key = v_event_key) then
      insert into public.stock_history (event_key, company_id, action, position, material_code, note, created_at_ms, expiry_date)
      values (
        v_event_key,
        p_company_id,
        'MOVE_ALL',
        p_source_position,
        v_row.material_code,
        coalesce(p_note, '') || ' | TARGET:' || p_target_position,
        v_created_at_ms,
        coalesce(v_row.expiry_date, '')
      );
    end if;

    insert into public.stock (company_id, position, material_code, quantity, expiry_date)
    values (p_company_id, p_target_position, v_row.material_code, v_row.quantity, coalesce(v_row.expiry_date, ''))
    on conflict (company_id, position, material_code)
    do nothing;

    v_materials := v_materials + 1;
  end loop;

  delete from public.stock
  where company_id = p_company_id
    and position = p_source_position;

  return jsonb_build_object(
    'ok', true,
    'action', 'MOVE_ALL',
    'event_key_base', v_event_key_base,
    'materials', v_materials
  );
end;
$$;

create or replace function public.wms_rebuild_company_stock(p_company_id uuid)
returns void
language plpgsql
security definer
set search_path = public
as $$
begin
  perform public.wms_require_company_access(p_company_id);

  delete from public.stock
  where company_id = p_company_id;

  insert into public.stock (company_id, position, material_code, quantity, expiry_date)
  select
    h.company_id,
    h.position,
    h.material_code,
    sum(
      case h.action
        when 'RECEIVE' then h.quantity
        when 'ISSUE' then -h.quantity
        when 'UNDO' then 0
        when 'ADJUST' then 0
        when 'MOVE_ALL' then 0
        else 0
      end
    )::int as quantity,
    coalesce(
      max(nullif(h.expiry_date, '')) filter (where h.action in ('RECEIVE', 'UNDO', 'ADJUST')),
      ''
    ) as expiry_date
  from public.stock_history h
  where h.company_id = p_company_id
  group by h.company_id, h.position, h.material_code
  having sum(
    case h.action
      when 'RECEIVE' then h.quantity
      when 'ISSUE' then -h.quantity
      when 'UNDO' then 0
      when 'ADJUST' then 0
      when 'MOVE_ALL' then 0
      else 0
    end
  ) > 0;
end;
$$;

grant execute on function public.wms_rebuild_company_stock(uuid) to authenticated;

create or replace function public.wms_receive_material(
  p_company_id uuid,
  p_position text,
  p_material_code text,
  p_note text default '',
  p_expiry_date text default '',
  p_quantity integer default 1,
  p_created_at_ms bigint default null,
  p_event_key text default null
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_event_key text := coalesce(nullif(btrim(p_event_key), ''), gen_random_uuid()::text);
  v_created_at_ms bigint := coalesce(p_created_at_ms, floor(extract(epoch from clock_timestamp()) * 1000)::bigint);
  v_existing_position text;
  v_existing_quantity integer;
  v_expiry_date text := btrim(coalesce(p_expiry_date, ''));
  v_quantity integer := coalesce(p_quantity, 0);
begin
  perform public.wms_require_company_access(p_company_id);

  if btrim(coalesce(p_position, '')) = '' then
    raise exception 'Position is required';
  end if;

  if btrim(coalesce(p_material_code, '')) = '' then
    raise exception 'Material code is required';
  end if;

  if v_quantity <= 0 then
    raise exception 'Quantity must be greater than 0';
  end if;

  if exists (select 1 from public.stock_history where event_key = v_event_key) then
    return jsonb_build_object(
      'ok', true,
      'duplicate', true,
      'event_key', v_event_key,
      'action', 'RECEIVE'
    );
  end if;

  select position, quantity
    into v_existing_position, v_existing_quantity
  from public.stock
  where company_id = p_company_id
    and material_code = p_material_code
  limit 1
  for update;

  if v_existing_position is not null and v_existing_position <> p_position then
    raise exception 'Material % already exists at position %', p_material_code, v_existing_position;
  end if;

  insert into public.stock_history (
    event_key,
    company_id,
    action,
    position,
    material_code,
    note,
    created_at_ms,
    expiry_date,
    quantity
  )
  values (
    v_event_key,
    p_company_id,
    'RECEIVE',
    p_position,
    p_material_code,
    coalesce(p_note, ''),
    v_created_at_ms,
    v_expiry_date,
    v_quantity
  );

  if v_existing_position is null then
    insert into public.stock (company_id, position, material_code, quantity, expiry_date)
    values (p_company_id, p_position, p_material_code, v_quantity, v_expiry_date);
  else
    update public.stock
    set quantity = coalesce(v_existing_quantity, 0) + v_quantity,
        expiry_date = case
          when v_expiry_date <> '' then v_expiry_date
          else coalesce(expiry_date, '')
        end
    where company_id = p_company_id
      and position = p_position
      and material_code = p_material_code;
  end if;

  return jsonb_build_object(
    'ok', true,
    'event_key', v_event_key,
    'action', 'RECEIVE',
    'quantity', v_quantity
  );
end;
$$;

create or replace function public.wms_issue_material(
  p_company_id uuid,
  p_position text,
  p_material_code text,
  p_note text default '',
  p_quantity integer default 1,
  p_created_at_ms bigint default null,
  p_event_key text default null
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_event_key text := coalesce(nullif(btrim(p_event_key), ''), gen_random_uuid()::text);
  v_created_at_ms bigint := coalesce(p_created_at_ms, floor(extract(epoch from clock_timestamp()) * 1000)::bigint);
  v_qty integer;
  v_expiry_date text;
  v_quantity integer := coalesce(p_quantity, 0);
begin
  perform public.wms_require_company_access(p_company_id);

  if v_quantity <= 0 then
    raise exception 'Quantity must be greater than 0';
  end if;

  if exists (select 1 from public.stock_history where event_key = v_event_key) then
    return jsonb_build_object(
      'ok', true,
      'duplicate', true,
      'event_key', v_event_key,
      'action', 'ISSUE'
    );
  end if;

  select quantity, expiry_date
    into v_qty, v_expiry_date
  from public.stock
  where company_id = p_company_id
    and position = p_position
    and material_code = p_material_code
  for update;

  if coalesce(v_qty, 0) < v_quantity then
    raise exception 'Not enough stock for material % at position % (requested %, available %)',
      p_material_code,
      p_position,
      v_quantity,
      coalesce(v_qty, 0);
  end if;

  insert into public.stock_history (
    event_key,
    company_id,
    action,
    position,
    material_code,
    note,
    created_at_ms,
    expiry_date,
    quantity
  )
  values (
    v_event_key,
    p_company_id,
    'ISSUE',
    p_position,
    p_material_code,
    coalesce(p_note, ''),
    v_created_at_ms,
    coalesce(v_expiry_date, ''),
    v_quantity
  );

  if v_qty = v_quantity then
    delete from public.stock
    where company_id = p_company_id
      and position = p_position
      and material_code = p_material_code;
  else
    update public.stock
    set quantity = v_qty - v_quantity
    where company_id = p_company_id
      and position = p_position
      and material_code = p_material_code;
  end if;

  return jsonb_build_object(
    'ok', true,
    'event_key', v_event_key,
    'action', 'ISSUE',
    'quantity', v_quantity,
    'remaining_quantity', greatest(v_qty - v_quantity, 0)
  );
end;
$$;

create or replace function public.wms_move_all_position(
  p_company_id uuid,
  p_source_position text,
  p_target_position text,
  p_note text default '',
  p_created_at_ms bigint default null,
  p_event_key_base text default null
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_created_at_ms bigint := coalesce(p_created_at_ms, floor(extract(epoch from clock_timestamp()) * 1000)::bigint);
  v_event_key_base text := coalesce(nullif(btrim(p_event_key_base), ''), gen_random_uuid()::text);
  v_row record;
  v_event_key text;
  v_materials integer := 0;
begin
  perform public.wms_require_company_access(p_company_id);

  if p_source_position = p_target_position then
    raise exception 'Source and target position must be different';
  end if;

  for v_row in
    select material_code, quantity, expiry_date
    from public.stock
    where company_id = p_company_id
      and position = p_source_position
    for update
  loop
    if coalesce(v_row.quantity, 0) <= 0 then
      continue;
    end if;

    if exists (
      select 1
      from public.stock
      where company_id = p_company_id
        and position = p_target_position
        and material_code = v_row.material_code
    ) then
      raise exception 'Material % already exists at target position %', v_row.material_code, p_target_position;
    end if;

    v_event_key := v_event_key_base || ':' || v_row.material_code;

    if not exists (select 1 from public.stock_history where event_key = v_event_key) then
      insert into public.stock_history (
        event_key,
        company_id,
        action,
        position,
        material_code,
        note,
        created_at_ms,
        expiry_date,
        quantity
      )
      values (
        v_event_key,
        p_company_id,
        'MOVE_ALL',
        p_source_position,
        v_row.material_code,
        coalesce(p_note, '') || ' | TARGET:' || p_target_position,
        v_created_at_ms,
        coalesce(v_row.expiry_date, ''),
        v_row.quantity
      );
    end if;

    insert into public.stock (company_id, position, material_code, quantity, expiry_date)
    values (p_company_id, p_target_position, v_row.material_code, v_row.quantity, coalesce(v_row.expiry_date, ''))
    on conflict (company_id, position, material_code)
    do nothing;

    v_materials := v_materials + 1;
  end loop;

  delete from public.stock
  where company_id = p_company_id
    and position = p_source_position;

  return jsonb_build_object(
    'ok', true,
    'action', 'MOVE_ALL',
    'event_key_base', v_event_key_base,
    'materials', v_materials
  );
end;
$$;

grant execute on function public.wms_move_all_position(uuid, text, text, text, bigint, text) to authenticated;

create or replace function public.mes_factory_overview(p_company_id uuid default null)
returns table (
  workstation_id uuid,
  workstation_code text,
  workstation_name text,
  workstation_area text,
  workstation_active boolean,
  hmi_enabled boolean,
  machine_id uuid,
  machine_code text,
  machine_name text,
  machine_state text,
  machine_last_heartbeat_at timestamptz,
  terminal_id uuid,
  terminal_code text,
  terminal_name text,
  terminal_last_seen_at timestamptz,
  job_run_id uuid,
  production_order_id uuid,
  job_number text,
  item_code text,
  item_name text,
  job_status text,
  operator_name text,
  planned_quantity integer,
  good_quantity integer,
  scrap_quantity integer,
  started_at timestamptz,
  current_downtime_reason text,
  latest_event_at timestamptz
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_company_id uuid;
begin
  if p_company_id is null then
    v_company_id := public.user_company_id(auth.uid());
  else
    v_company_id := p_company_id;
  end if;

  if v_company_id is null then
    raise exception 'Company scope is required';
  end if;

  if not public.is_master(auth.uid()) and v_company_id <> public.user_company_id(auth.uid()) then
    raise exception 'Access denied for company %', v_company_id;
  end if;

  return query
  with active_runs as (
    select distinct on (jr.workstation_id)
      jr.id,
      jr.company_id,
      jr.production_order_id,
      jr.workstation_id,
      jr.machine_id,
      jr.terminal_id,
      jr.job_number,
      jr.item_code,
      jr.item_name,
      jr.status,
      jr.operator_name,
      jr.planned_quantity,
      jr.good_quantity,
      jr.scrap_quantity,
      jr.started_at,
      jr.created_at
    from public.mes_job_runs jr
    where jr.company_id = v_company_id
      and jr.status in ('queued', 'running', 'paused')
    order by
      jr.workstation_id,
      case jr.status when 'running' then 0 when 'paused' then 1 else 2 end,
      coalesce(jr.started_at, jr.created_at) desc
  ),
  last_events as (
    select distinct on (e.job_run_id)
      e.job_run_id,
      e.event_type,
      e.happened_at,
      dr.name as downtime_reason_name
    from public.mes_job_run_events e
    left join public.mes_downtime_reasons dr on dr.id = e.downtime_reason_id
    order by e.job_run_id, e.happened_at desc, e.created_at desc
  )
  select
    ws.id as workstation_id,
    ws.code as workstation_code,
    ws.name as workstation_name,
    ws.area as workstation_area,
    ws.is_active as workstation_active,
    ws.hmi_enabled,
    m.id as machine_id,
    m.code as machine_code,
    m.name as machine_name,
    m.machine_state,
    m.last_heartbeat_at as machine_last_heartbeat_at,
    t.id as terminal_id,
    t.terminal_code,
    t.name as terminal_name,
    t.last_seen_at as terminal_last_seen_at,
    ar.id as job_run_id,
    ar.production_order_id,
    ar.job_number,
    ar.item_code,
    ar.item_name,
    ar.status as job_status,
    ar.operator_name,
    ar.planned_quantity,
    ar.good_quantity,
    ar.scrap_quantity,
    ar.started_at,
    case when le.event_type = 'downtime_start' then le.downtime_reason_name else null end as current_downtime_reason,
    le.happened_at as latest_event_at
  from public.mes_workstations ws
  left join lateral (
    select mm.id, mm.code, mm.name, mm.machine_state, mm.last_heartbeat_at
    from public.mes_machines mm
    where mm.workstation_id = ws.id
      and mm.is_active = true
    order by mm.created_at asc
    limit 1
  ) m on true
  left join lateral (
    select tt.id, tt.terminal_code, tt.name, tt.last_seen_at
    from public.mes_hmi_terminals tt
    where tt.workstation_id = ws.id
      and tt.is_active = true
    order by tt.created_at asc
    limit 1
  ) t on true
  left join active_runs ar
    on ar.workstation_id = ws.id
  left join last_events le
    on le.job_run_id = ar.id
  where ws.company_id = v_company_id
  order by ws.sort_order asc, ws.name asc;
end;
$$;

create or replace function public.can_access_mes(uid uuid default auth.uid())
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1
    from public.app_users au
    left join public.companies c on c.id = au.company_id
    where au.user_id = uid
      and (
        au.role = 'master'
        or (
          au.can_access_mes = true
          and coalesce(c.mes_enabled, false) = true
        )
      )
  );
$$;

create or replace function public.set_company_mes_enabled(target_company_id uuid, target_enabled boolean)
returns public.companies
language plpgsql
security definer
set search_path = public
as $$
declare
  v_company public.companies;
begin
  if not public.is_master(auth.uid()) then
    raise exception 'Only master can change MES company access';
  end if;

  if target_company_id is null then
    raise exception 'company_id is required';
  end if;

  update public.companies
  set mes_enabled = coalesce(target_enabled, false)
  where id = target_company_id
  returning * into v_company;

  if v_company is null then
    raise exception 'company not found';
  end if;

  return v_company;
end;
$$;

create or replace function public.set_user_mes_access(target_user_id uuid, target_enabled boolean)
returns public.app_users
language plpgsql
security definer
set search_path = public
as $$
declare
  v_row public.app_users;
begin
  if not public.is_master(auth.uid()) then
    raise exception 'Only master can change MES user access';
  end if;

  if target_user_id is null then
    raise exception 'user_id is required';
  end if;

  update public.app_users
  set can_access_mes = case when role = 'master' then true else coalesce(target_enabled, false) end,
      updated_at = now()
  where user_id = target_user_id
  returning * into v_row;

  if v_row is null then
    raise exception 'app_user not found';
  end if;

  return v_row;
end;
$$;

create or replace function public.mes_factory_overview(p_company_id uuid default null)
returns table (
  workstation_id uuid,
  workstation_code text,
  workstation_name text,
  workstation_area text,
  workstation_active boolean,
  hmi_enabled boolean,
  machine_id uuid,
  machine_code text,
  machine_name text,
  machine_state text,
  machine_last_heartbeat_at timestamptz,
  terminal_id uuid,
  terminal_code text,
  terminal_name text,
  terminal_last_seen_at timestamptz,
  job_run_id uuid,
  production_order_id uuid,
  job_number text,
  item_code text,
  item_name text,
  job_status text,
  operator_name text,
  planned_quantity integer,
  good_quantity integer,
  scrap_quantity integer,
  started_at timestamptz,
  current_downtime_reason text,
  latest_event_at timestamptz
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_company_id uuid;
  v_user_company_id uuid;
begin
  v_user_company_id := public.user_company_id(auth.uid());

  if p_company_id is null then
    v_company_id := v_user_company_id;
  else
    v_company_id := p_company_id;
  end if;

  if v_company_id is null then
    raise exception 'Company scope is required';
  end if;

  if not public.is_master(auth.uid()) then
    if not public.can_access_mes(auth.uid()) then
      raise exception 'MES access denied';
    end if;

    if v_company_id <> v_user_company_id then
      raise exception 'Access denied for company %', v_company_id;
    end if;
  end if;

  return query
  with active_runs as (
    select distinct on (jr.workstation_id)
      jr.id,
      jr.company_id,
      jr.production_order_id,
      jr.workstation_id,
      jr.machine_id,
      jr.terminal_id,
      jr.job_number,
      jr.item_code,
      jr.item_name,
      jr.status,
      jr.operator_name,
      jr.planned_quantity,
      jr.good_quantity,
      jr.scrap_quantity,
      jr.started_at,
      jr.created_at
    from public.mes_job_runs jr
    where jr.company_id = v_company_id
      and jr.status in ('queued', 'running', 'paused')
    order by
      jr.workstation_id,
      case jr.status when 'running' then 0 when 'paused' then 1 else 2 end,
      coalesce(jr.started_at, jr.created_at) desc
  ),
  last_events as (
    select distinct on (e.job_run_id)
      e.job_run_id,
      e.event_type,
      e.happened_at,
      dr.name as downtime_reason_name
    from public.mes_job_run_events e
    left join public.mes_downtime_reasons dr on dr.id = e.downtime_reason_id
    order by e.job_run_id, e.happened_at desc, e.created_at desc
  )
  select
    ws.id as workstation_id,
    ws.code as workstation_code,
    ws.name as workstation_name,
    ws.area as workstation_area,
    ws.is_active as workstation_active,
    ws.hmi_enabled,
    m.id as machine_id,
    m.code as machine_code,
    m.name as machine_name,
    m.machine_state,
    m.last_heartbeat_at as machine_last_heartbeat_at,
    t.id as terminal_id,
    t.terminal_code,
    t.name as terminal_name,
    t.last_seen_at as terminal_last_seen_at,
    ar.id as job_run_id,
    ar.production_order_id,
    ar.job_number,
    ar.item_code,
    ar.item_name,
    ar.status as job_status,
    ar.operator_name,
    ar.planned_quantity,
    ar.good_quantity,
    ar.scrap_quantity,
    ar.started_at,
    case when le.event_type = 'downtime_start' then le.downtime_reason_name else null end as current_downtime_reason,
    le.happened_at as latest_event_at
  from public.mes_workstations ws
  left join lateral (
    select mm.id, mm.code, mm.name, mm.machine_state, mm.last_heartbeat_at
    from public.mes_machines mm
    where mm.workstation_id = ws.id
      and mm.is_active = true
    order by mm.created_at asc
    limit 1
  ) m on true
  left join lateral (
    select tt.id, tt.terminal_code, tt.name, tt.last_seen_at
    from public.mes_hmi_terminals tt
    where tt.workstation_id = ws.id
      and tt.is_active = true
    order by tt.created_at asc
    limit 1
  ) t on true
  left join active_runs ar
    on ar.workstation_id = ws.id
  left join last_events le
    on le.job_run_id = ar.id
  where ws.company_id = v_company_id
  order by ws.sort_order asc, ws.name asc;
end;
$$;

create or replace function public.bootstrap_self_signup(p_company_name text, p_username text default null)
returns jsonb
language plpgsql
security definer
set search_path = public, auth
as $$
declare
  v_user_id uuid := auth.uid();
  v_email text;
  v_existing public.app_users%rowtype;
  v_company_id uuid;
  v_username text;
begin
  if v_user_id is null then
    raise exception 'Musis byt prihlaseny.';
  end if;

  select *
  into v_existing
  from public.app_users
  where user_id = v_user_id
  limit 1;

  if v_existing.user_id is not null then
    return jsonb_build_object(
      'company_id', v_existing.company_id,
      'already_initialized', true
    );
  end if;

  select lower(coalesce(email, ''))
  into v_email
  from auth.users
  where id = v_user_id;

  if coalesce(v_email, '') = '' then
    raise exception 'K uctu sa nepodarilo nacitat email.';
  end if;

  if coalesce(trim(p_company_name), '') = '' then
    raise exception 'Nazov firmy nemoze byt prazdny.';
  end if;

  if exists (
    select 1
    from public.companies
    where lower(name) = lower(trim(p_company_name))
  ) then
    raise exception 'Firma s tymto nazvom uz existuje.';
  end if;

  v_username := lower(regexp_replace(coalesce(nullif(trim(p_username), ''), split_part(v_email, '@', 1)), '[^a-z0-9._-]+', '', 'g'));
  if coalesce(v_username, '') = '' then
    v_username := 'user-' || left(replace(v_user_id::text, '-', ''), 8);
  end if;

  if exists (
    select 1
    from public.app_users
    where lower(username) = lower(v_username)
      and user_id <> v_user_id
  ) then
    v_username := left(v_username, 22) || '-' || left(replace(v_user_id::text, '-', ''), 6);
  end if;

  insert into public.companies (name)
  values (trim(p_company_name))
  returning id into v_company_id;

  insert into public.company_profiles (
    company_id,
    name,
    invoice_due_days,
    invoice_style,
    created_by,
    updated_by
  )
  values (
    v_company_id,
    trim(p_company_name),
    14,
    'clean',
    v_user_id,
    v_user_id
  )
  on conflict (company_id) do update
    set name = excluded.name,
        updated_by = excluded.updated_by;

  insert into public.app_users (
    user_id,
    email,
    username,
    role,
    can_manage_orders,
    can_access_mes,
    company_id,
    created_by
  )
  values (
    v_user_id,
    v_email,
    v_username,
    'user',
    true,
    true,
    v_company_id,
    v_user_id
  )
  on conflict (user_id) do update
    set email = excluded.email,
        username = excluded.username,
        company_id = excluded.company_id,
        can_manage_orders = true,
        can_access_mes = true;

  return jsonb_build_object(
    'company_id', v_company_id,
    'company_name', trim(p_company_name),
    'username', v_username
  );
end;
$$;

grant execute on function public.bootstrap_self_signup(text, text) to authenticated;

create or replace function public.accept_company_invite(p_token text, p_username text default null)
returns jsonb
language plpgsql
security definer
set search_path = public, auth
as $$
declare
  v_user_id uuid := auth.uid();
  v_email text;
  v_existing public.app_users%rowtype;
  v_invite public.company_invites%rowtype;
  v_username text;
begin
  if v_user_id is null then
    raise exception 'Musis byt prihlaseny.';
  end if;

  select lower(coalesce(email, ''))
  into v_email
  from auth.users
  where id = v_user_id;

  if coalesce(v_email, '') = '' then
    raise exception 'K uctu sa nepodarilo nacitat email.';

  into v_invite

  end if;

  if v_invite.expires_at < now() then
    update public.company_invites
    set status = 'revoked',
        revoked_at = now()
    where id = v_invite.id;
    raise exception 'Pozvanka uz expirovala.';
  end if;

  if lower(v_invite.email) <> v_email then
    raise exception 'Pozvanka je viazana na iny email.';
  end if;

  select *
  into v_existing
  from public.app_users
  where user_id = v_user_id
  limit 1;

  if v_existing.user_id is not null and v_existing.role = 'master' then
    raise exception 'Master ucet nemozno pripojit cez firemnu pozvanku.';
  end if;

  if v_existing.user_id is not null and v_existing.company_id is not null and v_existing.company_id <> v_invite.company_id then
    raise exception 'Ucet uz patri do inej firmy.';
  end if;

  v_username := lower(regexp_replace(coalesce(nullif(trim(p_username), ''), nullif(v_existing.username, ''), split_part(v_email, '@', 1)), '[^a-z0-9._-]+', '', 'g'));
  if coalesce(v_username, '') = '' then
    v_username := 'user-' || left(replace(v_user_id::text, '-', ''), 8);
  end if;

  if exists (
    select 1
    from public.app_users
    where lower(username) = lower(v_username)
      and user_id <> v_user_id
  ) then
    v_username := left(v_username, 22) || '-' || left(replace(v_user_id::text, '-', ''), 6);
  end if;

  insert into public.app_users (
    user_id,
    email,
    username,
    role,
    can_manage_orders,
    can_access_mes,
    company_id,
    created_by
  )
  values (
    v_user_id,
    v_email,
    v_username,
    'user',
    v_invite.can_manage_orders,
    v_invite.can_access_mes,
    v_invite.company_id,
    coalesce(v_invite.created_by, v_user_id)
  )
  on conflict (user_id) do update
    set email = excluded.email,
        username = excluded.username,
        company_id = excluded.company_id,
        can_manage_orders = excluded.can_manage_orders,
        can_access_mes = excluded.can_access_mes;

  update public.company_invites
  set status = 'accepted',
      accepted_at = now(),
      accepted_by = v_user_id
  where id = v_invite.id;

  return jsonb_build_object(
    'company_id', v_invite.company_id,
    'email', v_email,
    'username', v_username
  );
end;
$$;
