-- MES terminal event log bridge for ESP32/HMI integrations.
-- Run after: 39_wms_mes_core.txt, 40_wms_mes_access_control.txt, 55_wms_mes_machine_signal_mode.txt

create extension if not exists pgcrypto;

create table if not exists public.mes_event_log (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  terminal_id uuid references public.mes_hmi_terminals (id) on delete set null,
  workstation_id uuid references public.mes_workstations (id) on delete set null,
  job_run_id uuid references public.mes_job_runs (id) on delete set null,
  terminal_event_id text not null,
  event_code text not null,
  job_number text not null default '',
  duration_seconds integer not null default 0 check (duration_seconds >= 0),
  time_from timestamptz,
  time_to timestamptz,
  operator_id text not null default '',
  downtime_reason_code text not null default '',
  downtime_reason_name text not null default '',
  payload jsonb not null default '{}'::jsonb,
  created_at timestamptz not null default now()
);

create unique index if not exists idx_mes_event_log_terminal_event_unique
  on public.mes_event_log (terminal_event_id);

create index if not exists idx_mes_event_log_company_created
  on public.mes_event_log (company_id, created_at desc);

create index if not exists idx_mes_event_log_terminal_created
  on public.mes_event_log (terminal_id, created_at desc);

create index if not exists idx_mes_event_log_workstation_created
  on public.mes_event_log (workstation_id, created_at desc);

create index if not exists idx_mes_event_log_job_run_created
  on public.mes_event_log (job_run_id, created_at desc);

alter table public.mes_event_log enable row level security;

drop policy if exists mes_event_log_select_auth on public.mes_event_log;

create policy mes_event_log_select_auth
on public.mes_event_log
for select
to authenticated
using (
  public.is_master(auth.uid())
  or (
    public.can_access_mes(auth.uid())
    and company_id = public.user_company_id(auth.uid())
  )
);

create or replace function public.mes_normalize_compact_event_code(p_event_code text, p_payload jsonb default '{}'::jsonb)
returns text
language sql
immutable
as $$
  select case lower(nullif(btrim(coalesce(p_payload->>'raw_event_type', '')), ''))
    when 'login' then 'ol'
    when 'logout' then 'oso'
    when 'start_operation' then 'start'
    when 'finish_operation' then 'of'
    when 'downtime_start' then 'stop'
    when 'downtime_end' then 'start'
    when 'machine_log' then 'ml'
    when 'barcode_scan' then 'os'
    when 'scrap_report' then 'scrap_count'
    else coalesce(
      nullif(lower(btrim(coalesce(p_payload->>'compact_event', ''))), ''),
      lower(btrim(coalesce(p_event_code, '')))
    )
  end;
$$;

create or replace function public.mes_compact_event_to_job_event_type(p_event_code text, p_payload jsonb default '{}'::jsonb)
returns text
language sql
immutable
as $$
  select case public.mes_normalize_compact_event_code(p_event_code, p_payload)
    when 'start' then 'start'
    when 'stop' then 'downtime_start'
    when 'ml' then 'note'
    when 'of' then 'complete'
    when 'good_count' then 'good_count'
    when 'scrap_count' then 'scrap_count'
    when 'ol' then 'note'
    when 'oso' then 'note'
    when 'os' then 'queue'
    else 'note'
  end;
$$;

create or replace function public.mes_log_compact_event_by_terminal_code(
  p_terminal_code text,
  p_terminal_event_id text,
  p_event_code text,
  p_happened_at timestamptz default null,
  p_job_number text default '',
  p_duration_seconds integer default 0,
  p_time_from timestamptz default null,
  p_time_to timestamptz default null,
  p_operator_id text default '',
  p_downtime_reason_code text default '',
  p_downtime_reason_name text default '',
  p_payload jsonb default '{}'::jsonb
)
returns table (
  event_id uuid,
  event_time timestamptz,
  event_code text,
  job_run_id uuid
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_terminal public.mes_hmi_terminals;
  v_company_id uuid;
  v_workstation_id uuid;
  v_machine_id uuid;
  v_production_order_id uuid;
  v_job_run_id uuid;
  v_downtime_reason_id uuid;
  v_event_id uuid;
  v_event_time timestamptz := coalesce(p_happened_at, now());
  v_event_code text := public.mes_normalize_compact_event_code(p_event_code, coalesce(p_payload, '{}'::jsonb));
  v_job_event_type text := public.mes_compact_event_to_job_event_type(p_event_code, coalesce(p_payload, '{}'::jsonb));
  v_job_number text := coalesce(
    nullif(btrim(p_job_number), ''),
    nullif(btrim(coalesce(p_payload->>'job_number', '')), ''),
    nullif(btrim(coalesce(p_payload->>'job', '')), ''),
    nullif(btrim(coalesce(p_payload->>'order_code', '')), ''),
    ''
  );
  v_quantity integer := greatest(
    0,
    case
      when coalesce(p_payload->>'quantity', '') ~ '^[0-9]+$' then (p_payload->>'quantity')::integer
      when coalesce(p_payload->>'qty', '') ~ '^[0-9]+$' then (p_payload->>'qty')::integer
      when coalesce(p_payload->>'count', '') ~ '^[0-9]+$' then (p_payload->>'count')::integer
      else 0
    end
  );
  v_payload jsonb := coalesce(p_payload, '{}'::jsonb);
begin
  if nullif(btrim(p_terminal_code), '') is null then
    raise exception 'terminal_code is required';
  end if;

  if nullif(btrim(p_terminal_event_id), '') is null then
    raise exception 'terminal_event_id is required';
  end if;

  select *
  into v_terminal
  from public.mes_hmi_terminals t
  where upper(btrim(t.terminal_code)) = upper(btrim(p_terminal_code))
  limit 1;

  if v_terminal.id is null then
    raise exception 'MES terminal % not found', p_terminal_code;
  end if;

  v_company_id := v_terminal.company_id;
  v_workstation_id := v_terminal.workstation_id;

  update public.mes_hmi_terminals
  set last_seen_at = v_event_time,
      app_version = coalesce(nullif(btrim(v_payload->>'app_version'), ''), app_version),
      last_ip = coalesce(nullif(btrim(v_payload->>'last_ip'), ''), last_ip)
  where id = v_terminal.id;

  select m.id
  into v_machine_id
  from public.mes_machines m
  where m.workstation_id = v_workstation_id
    and m.is_active = true
  order by m.created_at asc
  limit 1;

  if v_job_number <> '' and v_workstation_id is not null then
    select po.id
    into v_production_order_id
    from public.production_orders po
    where po.company_id = v_company_id
      and po.production_number = v_job_number
    limit 1;

    select jr.id
    into v_job_run_id
    from public.mes_job_runs jr
    where jr.company_id = v_company_id
      and jr.status in ('planned', 'queued', 'running', 'paused')
      and jr.job_number = v_job_number
      and (
        jr.terminal_id = v_terminal.id
        or jr.workstation_id = v_workstation_id
        or (v_machine_id is not null and jr.machine_id = v_machine_id)
      )
    order by
      case jr.status when 'running' then 0 when 'paused' then 1 when 'queued' then 2 else 3 end,
      coalesce(jr.started_at, jr.created_at) desc
    limit 1;

    if v_job_run_id is null then
      insert into public.mes_job_runs (
        company_id,
        production_order_id,
        workstation_id,
        machine_id,
        terminal_id,
        operator_name,
        job_number,
        item_code,
        item_name,
        status,
        started_at
      )
      values (
        v_company_id,
        v_production_order_id,
        v_workstation_id,
        v_machine_id,
        v_terminal.id,
        coalesce(nullif(btrim(p_operator_id), ''), nullif(btrim(v_payload->>'operator_id'), ''), ''),
        v_job_number,
        coalesce(nullif(btrim(v_payload->>'item_code'), ''), ''),
        coalesce(nullif(btrim(v_payload->>'item_name'), ''), ''),
        case when v_event_code = 'start' then 'running' else 'queued' end,
        case when v_event_code = 'start' then v_event_time else null end
      )
      returning id into v_job_run_id;
    end if;
  end if;

  if v_job_run_id is not null then
    update public.mes_job_runs
    set terminal_id = coalesce(terminal_id, v_terminal.id),
        machine_id = coalesce(machine_id, v_machine_id),
        production_order_id = coalesce(production_order_id, v_production_order_id),
        operator_name = coalesce(nullif(btrim(p_operator_id), ''), nullif(btrim(operator_name), ''), operator_name),
        status = case
          when v_event_code = 'start' then 'running'
          when v_event_code = 'stop' then 'paused'
          when v_event_code = 'of' then 'completed'
          else status
        end,
        started_at = case when v_event_code = 'start' then coalesce(started_at, v_event_time) else started_at end,
        ended_at = case when v_event_code = 'of' then coalesce(ended_at, v_event_time) else ended_at end,
        good_quantity = case when v_event_code = 'good_count' then good_quantity + greatest(v_quantity, 1) else good_quantity end,
        scrap_quantity = case when v_event_code = 'scrap_count' then scrap_quantity + greatest(v_quantity, 1) else scrap_quantity end,
        updated_at = now()
    where id = v_job_run_id;
  end if;

  if v_machine_id is not null then
    update public.mes_machines
    set machine_state = case
          when v_event_code = 'start' then 'running'
          when v_event_code in ('stop', 'ml') then 'down'
          when v_event_code = 'of' then 'idle'
          else machine_state
        end,
        last_heartbeat_at = v_event_time
    where id = v_machine_id;
  end if;

  insert into public.mes_event_log (
    company_id,
    terminal_id,
    workstation_id,
    job_run_id,
    terminal_event_id,
    event_code,
    job_number,
    duration_seconds,
    time_from,
    time_to,
    operator_id,
    downtime_reason_code,
    downtime_reason_name,
    payload,
    created_at
  )
  values (
    v_company_id,
    v_terminal.id,
    v_workstation_id,
    v_job_run_id,
    btrim(p_terminal_event_id),
    v_event_code,
    v_job_number,
    greatest(coalesce(p_duration_seconds, 0), 0),
    p_time_from,
    coalesce(p_time_to, v_event_time),
    coalesce(p_operator_id, ''),
    coalesce(p_downtime_reason_code, ''),
    coalesce(p_downtime_reason_name, ''),
    v_payload || jsonb_build_object('compact_event', v_event_code),
    v_event_time
  )
  on conflict (terminal_event_id) do update
  set event_code = excluded.event_code,
      job_run_id = coalesce(mes_event_log.job_run_id, excluded.job_run_id),
      payload = mes_event_log.payload || excluded.payload
  returning id into v_event_id;

  if v_job_run_id is not null and v_workstation_id is not null then
    select dr.id
    into v_downtime_reason_id
    from public.mes_downtime_reasons dr
    where dr.company_id = v_company_id
      and lower(dr.code) = lower(coalesce(p_downtime_reason_code, ''))
    limit 1;

    if not exists (
      select 1
      from public.mes_job_run_events e
      where e.payload->>'terminal_event_id' = btrim(p_terminal_event_id)
    ) then
      insert into public.mes_job_run_events (
        job_run_id,
        workstation_id,
        machine_id,
        downtime_reason_id,
        event_type,
        quantity,
        source,
        note,
        payload,
        happened_at
      )
      values (
        v_job_run_id,
        v_workstation_id,
        v_machine_id,
        v_downtime_reason_id,
        v_job_event_type,
        case when v_job_event_type in ('good_count', 'scrap_count') then greatest(v_quantity, 1) else 0 end,
        'hmi',
        coalesce(nullif(btrim(p_downtime_reason_name), ''), nullif(btrim(v_payload->>'note'), ''), ''),
        v_payload || jsonb_build_object('terminal_event_id', btrim(p_terminal_event_id), 'compact_event', v_event_code),
        v_event_time
      );
    end if;
  end if;

  return query select v_event_id, v_event_time, v_event_code, v_job_run_id;
end;
$$;

grant execute on function public.mes_normalize_compact_event_code(text, jsonb) to authenticated, anon, service_role;
grant execute on function public.mes_compact_event_to_job_event_type(text, jsonb) to authenticated, anon, service_role;
grant execute on function public.mes_log_compact_event_by_terminal_code(text, text, text, timestamptz, text, integer, timestamptz, timestamptz, text, text, text, jsonb) to authenticated, anon, service_role;
