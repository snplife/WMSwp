-- MES HMI terminal code normalization: accept P40-* as P4-*.

create or replace function public.mes_normalize_hmi_terminal_code()
returns trigger
language plpgsql
as $$
declare
  v_code text;
  v_suffix text;
begin
  v_code := upper(btrim(coalesce(new.terminal_code, '')));
  v_suffix := regexp_replace(v_code, '^(P40-|P4-|UID-)', '');

  if v_suffix ~ '^[0-9A-F]{12}$' then
    new.terminal_code := 'P4-' || v_suffix;
  else
    new.terminal_code := v_code;
  end if;

  return new;
end;
$$;

create or replace function public.mes_resolve_terminal_by_device_uid(p_device_uid text)
returns table (
  terminal_id uuid,
  company_id uuid,
  workstation_id uuid,
  terminal_code text,
  terminal_name text,
  is_active boolean,
  platform text,
  app_mode text,
  app_version text,
  last_ip text,
  last_seen_at timestamptz
)
language sql
security definer
set search_path = public
as $$
  with input_uid as (
    select
      upper(btrim(p_device_uid)) as raw_uid,
      regexp_replace(upper(btrim(p_device_uid)), '^(P40-|P4-|UID-)', '') as normalized_uid
  )
  select
    t.id as terminal_id,
    t.company_id,
    t.workstation_id,
    t.terminal_code,
    t.name as terminal_name,
    t.is_active,
    t.platform,
    t.app_mode,
    t.app_version,
    t.last_ip,
    t.last_seen_at
  from public.mes_hmi_terminals t
  cross join input_uid i
  where t.is_active is true
    and (
      upper(btrim(t.terminal_code)) = i.raw_uid
      or regexp_replace(upper(btrim(t.terminal_code)), '^(P40-|P4-|UID-)', '') = i.normalized_uid
    )
  order by
    case when upper(btrim(t.terminal_code)) = i.raw_uid then 0 else 1 end,
    t.last_seen_at desc nulls last,
    t.created_at desc
  limit 1;
$$;

grant execute on function public.mes_resolve_terminal_by_device_uid(text) to authenticated;
grant execute on function public.mes_resolve_terminal_by_device_uid(text) to anon;

update public.mes_hmi_terminals
set terminal_code = 'P4-' || regexp_replace(upper(btrim(terminal_code)), '^P40-', '')
where upper(btrim(terminal_code)) ~ '^P40-[0-9A-F]{12}$';
