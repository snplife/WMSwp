-- Add structured company registry fields to public.customers.

  on public.customers (company_id, ic_dph);

update public.customers
set
  ico = coalesce(
    nullif(ico, ''),
    nullif((regexp_match(note, '(?im)(?:^|\|)\s*ICO:\s*([^|\r\n]+)'))[1], '')
  ),
  dic = coalesce(
    nullif(dic, ''),
    nullif((regexp_match(note, '(?im)(?:^|\|)\s*DIC:\s*([^|\r\n]+)'))[1], '')
  ),
  ic_dph = coalesce(
    nullif(ic_dph, ''),
    nullif((regexp_match(note, '(?im)(?:^|\|)\s*IC DPH:\s*([^|\r\n]+)'))[1], '')
  )
where
  coalesce(nullif(ico, ''), nullif(dic, ''), nullif(ic_dph, '')) is null
  and note is not null
  and note <> '';

create or replace function public.wms_try_parse_invoice_meta(payload text)
returns jsonb
language plpgsql
as $$
begin
  return payload::jsonb;
exception
  when others then
    return null;
end;
$$;

with legacy as (
  select
    id,
    public.wms_try_parse_invoice_meta(substring(note from length('[[WMS_INVOICE_META]]') + 1)) as meta
  from public.invoices
  where note like '[[WMS_INVOICE_META]]%'
)
update public.invoices as i
set
  order_number = case
    when coalesce(i.order_number, '') = '' then coalesce(legacy.meta ->> 'orderNumber', '')
    else i.order_number
  end,
  intro_text = case
    when coalesce(i.intro_text, '') = '' then coalesce(legacy.meta ->> 'introText', '')
    else i.intro_text
  end,
  outro_text = case
    when coalesce(i.outro_text, '') = '' then coalesce(legacy.meta ->> 'outroText', '')
    else i.outro_text
  end,
  note = case
    when legacy.meta is not null then coalesce(legacy.meta ->> 'noteText', '')
    else i.note
  end
from legacy
where i.id = legacy.id;
