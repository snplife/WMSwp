-- Workstations = logical production workplaces shown on factory overview and bound to HMI terminals.
create table if not exists public.mes_workstations (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  code text not null,
  name text not null,
  area text not null default '',
  description text not null default '',
  target_cycle_seconds integer check (target_cycle_seconds is null or target_cycle_seconds > 0),
  ideal_units_per_hour numeric(12, 2) check (ideal_units_per_hour is null or ideal_units_per_hour > 0),
  hmi_enabled boolean not null default true,
  sort_order integer not null default 0,
  is_active boolean not null default true,
  created_by uuid references auth.users (id),
  updated_by uuid references auth.users (id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint mes_workstations_company_code_unique unique (company_id, code)
);

-- Machines = physical assets assigned to one workstation.
create table if not exists public.mes_machines (
  id uuid primary key default gen_random_uuid(),
  workstation_id uuid not null references public.mes_workstations (id) on delete cascade,
  code text not null,
  name text not null,
  asset_tag text not null default '',
  serial_number text not null default '',
  automation_mode text not null default 'full_automatic',

  last_heartbeat_at timestamptz,
  is_active boolean not null default true,
  created_by uuid references auth.users (id),
  updated_by uuid references auth.users (id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint mes_machines_workstation_code_unique unique (workstation_id, code)
);

  item_name text not null default '',

      'assign',
      'queue',
      'start',
      'pause',
      'resume',
      'downtime_start',
      'downtime_end',
      'good_count',
      'scrap_count',
      'setup_start',
      'setup_end',
      'complete',
      'cancel',
      'note'
    )),
  quantity integer not null default 0 check (quantity >= 0),
  source text not null default 'web'
    check (source in ('web', 'hmi', 'api', 'system')),
  note text not null default '',
  payload jsonb not null default '{}'::jsonb,
  happened_at timestamptz not null default now(),
  created_by uuid references auth.users (id) on delete set null,
  created_at timestamptz not null default now()
);

create or replace function public.mes_normalize_hmi_terminal_code()
returns trigger
language plpgsql
as $$
declare
  v_code text;
  v_suffix text;
begin
  v_code := upper(btrim(coalesce(new.terminal_code, '')));
  v_suffix := regexp_replace(v_code, '^(P40-|P4-|UID-)', '');

  if v_suffix ~ '^[0-9A-F]{12}$' then
    new.terminal_code := 'P4-' || v_suffix;
  else
    new.terminal_code := v_code;
  end if;

  return new;
end;
$$;

create or replace function public.mes_factory_overview(p_company_id uuid default null)
returns table (
  workstation_id uuid,
  workstation_code text,
  workstation_name text,
  workstation_area text,
  workstation_active boolean,
  hmi_enabled boolean,
  machine_id uuid,
  machine_code text,
  machine_name text,
  machine_state text,
  machine_last_heartbeat_at timestamptz,
  terminal_id uuid,
  terminal_code text,
  terminal_name text,
  terminal_last_seen_at timestamptz,
  job_run_id uuid,
  production_order_id uuid,
  job_number text,
  item_code text,
  item_name text,
  job_status text,
  operator_name text,
  planned_quantity integer,
  good_quantity integer,
  scrap_quantity integer,
  started_at timestamptz,
  current_downtime_reason text,
  latest_event_at timestamptz
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_company_id uuid;
begin
  if p_company_id is null then
    v_company_id := public.user_company_id(auth.uid());
  else
    v_company_id := p_company_id;
  end if;

  if v_company_id is null then
    raise exception 'Company scope is required';
  end if;

  if not public.is_master(auth.uid()) and v_company_id <> public.user_company_id(auth.uid()) then
    raise exception 'Access denied for company %', v_company_id;
  end if;

  return query
  with active_runs as (
    select distinct on (jr.workstation_id)
      jr.id,
      jr.company_id,
      jr.production_order_id,
      jr.workstation_id,
      jr.machine_id,
      jr.terminal_id,
      jr.job_number,
      jr.item_code,
      jr.item_name,
      jr.status,
      jr.operator_name,
      jr.planned_quantity,
      jr.good_quantity,
      jr.scrap_quantity,
      jr.started_at,
      jr.created_at
    from public.mes_job_runs jr
    where jr.company_id = v_company_id
      and jr.status in ('queued', 'running', 'paused')
    order by
      jr.workstation_id,
      case jr.status when 'running' then 0 when 'paused' then 1 else 2 end,
      coalesce(jr.started_at, jr.created_at) desc
  ),
  last_events as (
    select distinct on (e.job_run_id)
      e.job_run_id,
      e.event_type,
      e.happened_at,
      dr.name as downtime_reason_name
    from public.mes_job_run_events e
    left join public.mes_downtime_reasons dr on dr.id = e.downtime_reason_id
    order by e.job_run_id, e.happened_at desc, e.created_at desc
  )
  select
    ws.id as workstation_id,
    ws.code as workstation_code,
    ws.name as workstation_name,
    ws.area as workstation_area,
    ws.is_active as workstation_active,
    ws.hmi_enabled,
    m.id as machine_id,
    m.code as machine_code,
    m.name as machine_name,
    m.machine_state,
    m.last_heartbeat_at as machine_last_heartbeat_at,
    t.id as terminal_id,
    t.terminal_code,
    t.name as terminal_name,
    t.last_seen_at as terminal_last_seen_at,
    ar.id as job_run_id,
    ar.production_order_id,
    ar.job_number,
    ar.item_code,
    ar.item_name,
    ar.status as job_status,
    ar.operator_name,
    ar.planned_quantity,
    ar.good_quantity,
    ar.scrap_quantity,
    ar.started_at,
    case when le.event_type = 'downtime_start' then le.downtime_reason_name else null end as current_downtime_reason,
    le.happened_at as latest_event_at
  from public.mes_workstations ws
  left join lateral (
    select mm.id, mm.code, mm.name, mm.machine_state, mm.last_heartbeat_at
    from public.mes_machines mm
    where mm.workstation_id = ws.id
      and mm.is_active = true
    order by mm.created_at asc
    limit 1
  ) m on true
  left join lateral (
    select tt.id, tt.terminal_code, tt.name, tt.last_seen_at
    from public.mes_hmi_terminals tt
    where tt.workstation_id = ws.id
      and tt.is_active = true
    order by tt.created_at asc
    limit 1
  ) t on true
  left join active_runs ar
    on ar.workstation_id = ws.id
  left join last_events le
    on le.job_run_id = ar.id
  where ws.company_id = v_company_id
  order by ws.sort_order asc, ws.name asc;
end;
$$;

create or replace function public.can_access_mes(uid uuid default auth.uid())
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1
    from public.app_users au
    left join public.companies c on c.id = au.company_id
    where au.user_id = uid
      and (
        au.role = 'master'
        or (
          au.can_access_mes = true
          and coalesce(c.mes_enabled, false) = true
        )
      )
  );
$$;

create or replace function public.set_company_mes_enabled(target_company_id uuid, target_enabled boolean)
returns public.companies
language plpgsql
security definer
set search_path = public
as $$
declare
  v_company public.companies;
begin
  if not public.is_master(auth.uid()) then
    raise exception 'Only master can change MES company access';
  end if;

  if target_company_id is null then
    raise exception 'company_id is required';
  end if;

  update public.companies
  set mes_enabled = coalesce(target_enabled, false)
  where id = target_company_id
  returning * into v_company;

  if v_company is null then
    raise exception 'company not found';
  end if;

  return v_company;
end;
$$;

create or replace function public.set_user_mes_access(target_user_id uuid, target_enabled boolean)
returns public.app_users
language plpgsql
security definer
set search_path = public
as $$
declare
  v_row public.app_users;
begin
  if not public.is_master(auth.uid()) then
    raise exception 'Only master can change MES user access';
  end if;

  if target_user_id is null then
    raise exception 'user_id is required';
  end if;

  update public.app_users
  set can_access_mes = case when role = 'master' then true else coalesce(target_enabled, false) end,
      updated_at = now()
  where user_id = target_user_id
  returning * into v_row;

  if v_row is null then
    raise exception 'app_user not found';
  end if;

  return v_row;
end;
$$;

create or replace function public.mes_factory_overview(p_company_id uuid default null)
returns table (
  workstation_id uuid,
  workstation_code text,
  workstation_name text,
  workstation_area text,
  workstation_active boolean,
  hmi_enabled boolean,
  machine_id uuid,
  machine_code text,
  machine_name text,
  machine_state text,
  machine_last_heartbeat_at timestamptz,
  terminal_id uuid,
  terminal_code text,
  terminal_name text,
  terminal_last_seen_at timestamptz,
  job_run_id uuid,
  production_order_id uuid,
  job_number text,
  item_code text,
  item_name text,
  job_status text,
  operator_name text,
  planned_quantity integer,
  good_quantity integer,
  scrap_quantity integer,
  started_at timestamptz,
  current_downtime_reason text,
  latest_event_at timestamptz
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_company_id uuid;
  v_user_company_id uuid;
begin
  v_user_company_id := public.user_company_id(auth.uid());

  if p_company_id is null then
    v_company_id := v_user_company_id;
  else
    v_company_id := p_company_id;
  end if;

  if v_company_id is null then
    raise exception 'Company scope is required';
  end if;

  if not public.is_master(auth.uid()) then
    if not public.can_access_mes(auth.uid()) then
      raise exception 'MES access denied';
    end if;

    if v_company_id <> v_user_company_id then
      raise exception 'Access denied for company %', v_company_id;
    end if;
  end if;

  return query
  with active_runs as (
    select distinct on (jr.workstation_id)
      jr.id,
      jr.company_id,
      jr.production_order_id,
      jr.workstation_id,
      jr.machine_id,
      jr.terminal_id,
      jr.job_number,
      jr.item_code,
      jr.item_name,
      jr.status,
      jr.operator_name,
      jr.planned_quantity,
      jr.good_quantity,
      jr.scrap_quantity,
      jr.started_at,
      jr.created_at
    from public.mes_job_runs jr
    where jr.company_id = v_company_id
      and jr.status in ('queued', 'running', 'paused')
    order by
      jr.workstation_id,
      case jr.status when 'running' then 0 when 'paused' then 1 else 2 end,
      coalesce(jr.started_at, jr.created_at) desc
  ),
  last_events as (
    select distinct on (e.job_run_id)
      e.job_run_id,
      e.event_type,
      e.happened_at,
      dr.name as downtime_reason_name
    from public.mes_job_run_events e
    left join public.mes_downtime_reasons dr on dr.id = e.downtime_reason_id
    order by e.job_run_id, e.happened_at desc, e.created_at desc
  )
  select
    ws.id as workstation_id,
    ws.code as workstation_code,
    ws.name as workstation_name,
    ws.area as workstation_area,
    ws.is_active as workstation_active,
    ws.hmi_enabled,
    m.id as machine_id,
    m.code as machine_code,
    m.name as machine_name,
    m.machine_state,
    m.last_heartbeat_at as machine_last_heartbeat_at,
    t.id as terminal_id,
    t.terminal_code,
    t.name as terminal_name,
    t.last_seen_at as terminal_last_seen_at,
    ar.id as job_run_id,
    ar.production_order_id,
    ar.job_number,
    ar.item_code,
    ar.item_name,
    ar.status as job_status,
    ar.operator_name,
    ar.planned_quantity,
    ar.good_quantity,
    ar.scrap_quantity,
    ar.started_at,
    case when le.event_type = 'downtime_start' then le.downtime_reason_name else null end as current_downtime_reason,
    le.happened_at as latest_event_at
  from public.mes_workstations ws
  left join lateral (
    select mm.id, mm.code, mm.name, mm.machine_state, mm.last_heartbeat_at
    from public.mes_machines mm
    where mm.workstation_id = ws.id
      and mm.is_active = true
    order by mm.created_at asc
    limit 1
  ) m on true
  left join lateral (
    select tt.id, tt.terminal_code, tt.name, tt.last_seen_at
    from public.mes_hmi_terminals tt
    where tt.workstation_id = ws.id
      and tt.is_active = true
    order by tt.created_at asc
    limit 1
  ) t on true
  left join active_runs ar
    on ar.workstation_id = ws.id
  left join last_events le
    on le.job_run_id = ar.id
  where ws.company_id = v_company_id
  order by ws.sort_order asc, ws.name asc;
end;
$$;

create or replace function public.mes_resolve_terminal_by_device_uid(p_device_uid text)
returns table (
  terminal_id uuid,
  company_id uuid,
  workstation_id uuid,
  terminal_code text,
  terminal_name text,
  is_active boolean,
  platform text,
  app_mode text,
  app_version text,
  last_ip text,
  last_seen_at timestamptz
)
language sql
security definer
set search_path = public
as $$
  with input_uid as (
    select
      upper(btrim(p_device_uid)) as raw_uid,
      regexp_replace(upper(btrim(p_device_uid)), '^(P40-|P4-|UID-)', '') as normalized_uid
  )
  select
    t.id as terminal_id,
    t.company_id,
    t.workstation_id,
    t.terminal_code,
    t.name as terminal_name,
    t.is_active,
    t.platform,
    t.app_mode,
    t.app_version,
    t.last_ip,
    t.last_seen_at
  from public.mes_hmi_terminals t
  cross join input_uid i
  where t.is_active is true
    and (
      upper(btrim(t.terminal_code)) = i.raw_uid
      or regexp_replace(upper(btrim(t.terminal_code)), '^(P40-|P4-|UID-)', '') = i.normalized_uid
    )
  order by
    case when upper(btrim(t.terminal_code)) = i.raw_uid then 0 else 1 end,
    t.last_seen_at desc nulls last,
    t.created_at desc
  limit 1;
$$;
